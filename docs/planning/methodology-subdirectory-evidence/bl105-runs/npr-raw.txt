#!/bin/bash
# BL-105 rehearsal, nprcgenekeepr arms: serial queue (one heavy job at a time)
export RENV_PATHS_LIBRARY=$HOME/Library/Caches/org.R-project.R/R/renv/library/nprcgenekeepr-e7e04aee
R=/Users/rmsharp/.cache/bl105-rehearsal; E=/Users/rmsharp/Development/methodology/docs/planning/methodology-subdirectory-evidence
cd $R
for n in npr-base npr-move npr-edit; do
  ( cd $R/$n && timeout 900 Rscript vignettes/articles/data-raw/build-document1-evidence.R > $R/out/script-$n.txt 2>&1; echo $? > $R/out/script-$n.done; git status --short | head -20 > $R/out/script-$n.status.txt )
done
for n in npr-base npr-move; do
  ( cd $R/$n && python3 methodology/quality_ratchet.py --run > $R/out/gate-$n.txt 2>&1; echo $? > $R/out/gate-$n.done )
done
for n in npr-base npr-move; do
  bash $E/p15-r-run.sh $R/$n $E/p15-r-full.R $R/out/full-$n $R/out/full-$n.csv
done
echo finished > $R/out/queue-npr.done
